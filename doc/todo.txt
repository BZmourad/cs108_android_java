    private void updateImmo() {
        if (spinnerEtat.getSelectedItem() == null || spinnerEtat.getSelectedItemPosition() < 0) {
            Toast.makeText(getContext(), "Veuillez sélectionner un état", Toast.LENGTH_SHORT).show();
            return;
        }
        
        if (spinnerEmplacement.getSelectedItem() == null || spinnerEmplacement.getSelectedItemPosition() < 0) {
            Toast.makeText(getContext(), "Veuillez sélectionner un emplacement", Toast.LENGTH_SHORT).show();
            return;
        }
        
        showProgress(true);
        buttonUpdate.setEnabled(false);
        
        final String username = SharedPrefManager.getInstance(getContext()).getUsername();
        final String codeBarre = editTextCodeBarre.getText().toString().trim();
        final String numSerie = editTextNumSerie.getText().toString().trim();
        final String note = editTextNote.getText().toString().trim();
        final String codeEtat = etatCodeList.get(spinnerEtat.getSelectedItemPosition());
        final String codeEmp = emplacementCodeList.get(spinnerEmplacement.getSelectedItemPosition());
        
        MainActivity.csLibrary4A.appendToLogView("EditImmoFragment: Mise à jour immo - code_immo=" + codeImmo);
        
        new Thread(new Runnable() {
            @Override
            public void run() {
                HttpURLConnection connection = null;
                try {
                    URL url = new URL(baseUrl + "api-update-det-inv.awp?log=" + token + "&inv=" + codeInv + "&immo=" + codeImmo);
                    connection = (HttpURLConnection) url.openConnection();
                    connection.setRequestMethod("POST");
                    connection.setRequestProperty("Content-Type", "application/json");
                    connection.setDoOutput(true);
                    connection.setConnectTimeout(15000);
                    connection.setReadTimeout(15000);
                    
                    JSONObject jsonParam = new JSONObject();
                    jsonParam.put("user", username);
                    jsonParam.put("note", note);
                    jsonParam.put("code_barre", codeBarre);
                    jsonParam.put("num_serie", numSerie);
                    jsonParam.put("code_etat", codeEtat);
                    jsonParam.put("code_emp", codeEmp);
                    jsonParam.put("mission", codeMission);
                    jsonParam.put("equipe", equipe);
                    // Note: Code_RFID n'est plus envoyé
                    
                    MainActivity.csLibrary4A.appendToLogView("EditImmoFragment: Envoi données: " + jsonParam.toString());
                    
                    OutputStream os = connection.getOutputStream();
                    os.write(jsonParam.toString().getBytes("UTF-8"));
                    os.flush();
                    os.close();
                    
                    int responseCode = connection.getResponseCode();
                    MainActivity.csLibrary4A.appendToLogView("EditImmoFragment: Response code=" + responseCode);
                    
                    BufferedReader reader;
                    if (responseCode == HttpURLConnection.HTTP_OK) {
                        reader = new BufferedReader(new InputStreamReader(connection.getInputStream()));
                    } else {
                        reader = new BufferedReader(new InputStreamReader(connection.getErrorStream()));
                    }
                    
                    StringBuilder response = new StringBuilder();
                    String line;
                    while ((line = reader.readLine()) != null) {
                        response.append(line);
                    }
                    reader.close();
                    
                    MainActivity.csLibrary4A.appendToLogView("EditImmoFragment: Réponse=" + response.toString());
                    
                    JSONObject result = new JSONObject(response.toString());
                    if ("ok".equals(result.optString("sRes"))) {
                        getActivity().runOnUiThread(new Runnable() {
                            @Override
                            public void run() {
                                Toast.makeText(getContext(), "Immobilisation modifiée avec succès", Toast.LENGTH_LONG).show();
                                MainActivity.csLibrary4A.appendToLogView("EditImmoFragment: Mise à jour réussie");
                                showProgress(false);
                                getActivity().getSupportFragmentManager().popBackStack();
                            }
                        });
                    } else {
                        final String errorMsg = result.optString("sRes", "Erreur inconnue");
                        getActivity().runOnUiThread(new Runnable() {
                            @Override
                            public void run() {
                                showProgress(false);
                                Toast.makeText(getContext(), "Erreur: " + errorMsg, Toast.LENGTH_LONG).show();
                                MainActivity.csLibrary4A.appendToLogView("EditImmoFragment: Erreur réponse=" + errorMsg);
                            }
                        });
                    }
                    
                } catch (Exception e) {
                    e.printStackTrace();
                    MainActivity.csLibrary4A.appendToLogView("EditImmoFragment: Exception=" + e.getMessage());
                    getActivity().runOnUiThread(new Runnable() {
                        @Override
                        public void run() {
                            showProgress(false);
                            Toast.makeText(getContext(), "Erreur de connexion: " + e.getMessage(), Toast.LENGTH_LONG).show();
                        }
                    });
                } finally {
                    if (connection != null) {
                        connection.disconnect();
                    }
                }
            }
        }).start();
    }

package com.csl.cs108ademoapp.apps;
import android.content.Context;
import android.content.SharedPreferences;
public class SharedPrefManager {
    private static final String SHARED_PREF_NAME = "user_session";
    private static final String SHARED_PREF_CONFIG = "app_config";
    private static final String KEY_USERNAME = "key_username";
    private static final String KEY_MODE = "key_mode"; // online or offline
    private static final String KEY_TOKEN = "key_token"; // for online mode
    private static final String KEY_USER_CODE = "key_user_code";
    private static final String KEY_USER_NAME = "key_user_name";
    
    // Configuration keys
    private static final String KEY_SERVER_URL = "key_server_url";
    private static final String KEY_APP_MODE = "key_app_mode"; // "online" or "offline"
    
    private static SharedPrefManager mInstance;
    private static Context mCtx;

    private SharedPrefManager(Context context) {
        mCtx = context;
    }

    public static synchronized SharedPrefManager getInstance(Context context) {
        if (mInstance == null) {
            mInstance = new SharedPrefManager(context);
        }
        return mInstance;
    }

    public void userLogin(String username, boolean isOnline, String token, String code, String name) {
        SharedPreferences sharedPreferences = mCtx.getSharedPreferences(SHARED_PREF_NAME, Context.MODE_PRIVATE);
        SharedPreferences.Editor editor = sharedPreferences.edit();
        editor.putString(KEY_USERNAME, username);
        editor.putBoolean(KEY_MODE, isOnline);
        editor.putString(KEY_TOKEN, token);
        editor.putString(KEY_USER_CODE, code);
        editor.putString(KEY_USER_NAME, name);
        editor.apply();
    }

    public boolean isLoggedIn() {
        SharedPreferences sharedPreferences = mCtx.getSharedPreferences(SHARED_PREF_NAME, Context.MODE_PRIVATE);
        return sharedPreferences.getString(KEY_USERNAME, null) != null;
    }

    public String getUsername() {
        SharedPreferences sharedPreferences = mCtx.getSharedPreferences(SHARED_PREF_NAME, Context.MODE_PRIVATE);
        return sharedPreferences.getString(KEY_USERNAME, null);
    }

    public String getUserCode() {
        SharedPreferences sharedPreferences = mCtx.getSharedPreferences(SHARED_PREF_NAME, Context.MODE_PRIVATE);
        return sharedPreferences.getString(KEY_USER_CODE, null);
    }

    public String getUserName() {
        SharedPreferences sharedPreferences = mCtx.getSharedPreferences(SHARED_PREF_NAME, Context.MODE_PRIVATE);
        return sharedPreferences.getString(KEY_USER_NAME, null);
    }

    public boolean isOnlineMode() {
        SharedPreferences sharedPreferences = mCtx.getSharedPreferences(SHARED_PREF_NAME, Context.MODE_PRIVATE);
        return sharedPreferences.getBoolean(KEY_MODE, true); // default to online
    }

    public String getToken() {
        SharedPreferences sharedPreferences = mCtx.getSharedPreferences(SHARED_PREF_NAME, Context.MODE_PRIVATE);
        return sharedPreferences.getString(KEY_TOKEN, null);
    }

    public void logout() {
        SharedPreferences sharedPreferences = mCtx.getSharedPreferences(SHARED_PREF_NAME, Context.MODE_PRIVATE);
        SharedPreferences.Editor editor = sharedPreferences.edit();
        editor.clear();
        editor.apply();
    }
    
    // Configuration methods
    public void saveServerConfig(String serverUrl, String appMode) {
        SharedPreferences sharedPreferences = mCtx.getSharedPreferences(SHARED_PREF_CONFIG, Context.MODE_PRIVATE);
        SharedPreferences.Editor editor = sharedPreferences.edit();
        editor.putString(KEY_SERVER_URL, serverUrl);
        editor.putString(KEY_APP_MODE, appMode);
        editor.apply();
    }
    
    public String getServerUrl() {
        SharedPreferences sharedPreferences = mCtx.getSharedPreferences(SHARED_PREF_CONFIG, Context.MODE_PRIVATE);
        return sharedPreferences.getString(KEY_SERVER_URL, "http://196.203.88.132/GESTION_INV_CNAM_WEB/FR/");
    }
    
    public String getAppMode() {
        SharedPreferences sharedPreferences = mCtx.getSharedPreferences(SHARED_PREF_CONFIG, Context.MODE_PRIVATE);
        return sharedPreferences.getString(KEY_APP_MODE, "online");
    }
}

    public boolean addUser(Utilisateur utilisateur) {
        SQLiteDatabase db = this.getWritableDatabase();
        ContentValues values = new ContentValues();
        
        values.put("CodeUtil", utilisateur.CodeUtil);
        values.put("NomPrenomUtil", utilisateur.NomPrenomUtil);
        values.put("MotPasse", utilisateur.MotPasse);
        values.put("login", utilisateur.login);
        
        long result = db.insert(TABLE_UTILISATEUR, null, values);
        db.close();
        
        return result != -1;
    }

    public boolean checkUser(String login, String password) {
        SQLiteDatabase db = this.getReadableDatabase();
        String[] columns = {"id"};
        String selection = "login = ? AND MotPasse = ?";
        String[] selectionArgs = {login, password};
        
        Cursor cursor = db.query(TABLE_UTILISATEUR, columns, selection, selectionArgs, null, null, null);
        int count = cursor.getCount();
        cursor.close();
        db.close();
        
        return count > 0;
    }

    public Utilisateur getUserDetails(String login) {
        SQLiteDatabase db = this.getReadableDatabase();
        String[] columns = {"CodeUtil", "NomPrenomUtil", "MotPasse", "login"};
        String selection = "login = ?";
        String[] selectionArgs = {login};
        
        Cursor cursor = db.query(TABLE_UTILISATEUR, columns, selection, selectionArgs, null, null, null);
        
        Utilisateur utilisateur = null;
        if (cursor.moveToFirst()) {
            utilisateur = new Utilisateur();
            utilisateur.CodeUtil = cursor.getString(0);
            utilisateur.NomPrenomUtil = cursor.getString(1);
            utilisateur.MotPasse = cursor.getString(2);
            utilisateur.login = cursor.getString(3);
        }
        cursor.close();
        db.close();
        
        return utilisateur;
    }
